cmake_minimum_required(VERSION 3.16)
project(fast_livo LANGUAGES C CXX)
get_filename_component(FAST_LIVO2_ENTRY_FILE "${CMAKE_CURRENT_LIST_FILE}" REALPATH)
get_filename_component(FAST_LIVO2_ENTRY "${FAST_LIVO2_ENTRY_FILE}" DIRECTORY)
get_filename_component(FAST_LIVO2_ROOT "${FAST_LIVO2_ENTRY}/.." ABSOLUTE)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
find_package(catkin REQUIRED COMPONENTS roscpp livox_ros_driver sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros pcl_conversions cv_bridge visualization_msgs)
include(${FAST_LIVO2_ROOT}/cmake/fast_livo2_core.cmake)
catkin_package(LIBRARIES fast_livo2_core)
add_executable(fastlivo_mapping ${FAST_LIVO2_ROOT}/noetic/src/node.cpp)
target_include_directories(fastlivo_mapping PRIVATE ${FAST_LIVO2_ROOT}/adapters ${catkin_INCLUDE_DIRS})
target_link_libraries(fastlivo_mapping fast_livo2_core ${catkin_LIBRARIES})
add_dependencies(fastlivo_mapping ${catkin_EXPORTED_TARGETS})
install(TARGETS fastlivo_mapping RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY ${FAST_LIVO2_ROOT}/config ${FAST_LIVO2_ROOT}/rviz_cfg ${FAST_LIVO2_ROOT}/noetic/launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

if(CATKIN_ENABLE_TESTING)
  enable_testing()
  include(${FAST_LIVO2_ROOT}/cmake/fast_livo2_tests.cmake)
endif()
