cmake_minimum_required(VERSION 3.16)
project(fast_livo LANGUAGES C CXX)
get_filename_component(FAST_LIVO2_ENTRY_FILE "${CMAKE_CURRENT_LIST_FILE}" REALPATH)
get_filename_component(FAST_LIVO2_ENTRY "${FAST_LIVO2_ENTRY_FILE}" DIRECTORY)
get_filename_component(FAST_LIVO2_ROOT "${FAST_LIVO2_ENTRY}/.." ABSOLUTE)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(visualization_msgs REQUIRED)
include(${FAST_LIVO2_ROOT}/cmake/fast_livo2_core.cmake)
add_executable(fastlivo_mapping ${FAST_LIVO2_ROOT}/humble/src/node.cpp)
target_include_directories(fastlivo_mapping PRIVATE ${FAST_LIVO2_ROOT}/adapters)
target_link_libraries(fastlivo_mapping fast_livo2_core)
ament_target_dependencies(fastlivo_mapping rclcpp livox_ros_driver2 sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros pcl_conversions cv_bridge visualization_msgs)
install(TARGETS fastlivo_mapping DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY ${FAST_LIVO2_ROOT}/humble/launch ${FAST_LIVO2_ROOT}/config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  include(${FAST_LIVO2_ROOT}/cmake/fast_livo2_tests.cmake)
endif()
ament_package()
